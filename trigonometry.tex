\section*{Основные тождества}

\begin{enumerate}
    \item $\sin^2 x + \cos^2 x = 1$
    \item $1 + \tan^2 x = \sec^2 x$
    \item $1 + \cot^2 x = \csc^2 x$
\end{enumerate}

\section*{Формулы сложения}

\begin{enumerate}
    \item $\sin(x \pm y) = \sin x \cos y \pm \cos x \sin y$
    \item $\cos(x \pm y) = \cos x \cos y \mp \sin x \sin y$
\end{enumerate}

\section*{Формулы двойного угла}

\begin{enumerate}
    \item $\sin 2x = 2 \sin x \cos x$
    \item $\cos 2x = \cos^2 x - \sin^2 x = 2 \cos^2 x - 1 = 1 - 2 \sin^2 x$
\end{enumerate}
