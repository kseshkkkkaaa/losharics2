    \Huge\textbf{Шпаргалка}
\end{center}

\section*{Производные}
\begin{itemize}
    \item $(x^n)' = n \cdot x^{n-1}$
    \item $(\sin x)' = \cos x$
    \item $(\cos x)' = -\sin x$
    \item $(e^x)' = e^x$
    \item $(\ln x)' = \dfrac{1}{x}$
\end{itemize}

\section*{Интегралы}
\begin{itemize}
    \item $\int x^n \, dx = \dfrac{x^{n+1}}{n+1} + C \quad (n \neq -1)$
    \item $\int \dfrac{1}{x} \, dx = \ln|x| + C$
    \item $\int e^x \, dx = e^x + C$
    \item $\int \sin x \, dx = -\cos x + C$
    \item $\int \cos x \, dx = \sin x + C$
\end{itemize}

\section*{Пределы}
\begin{itemize}
    \item $\lim_{x \to 0} \dfrac{\sin x}{x} = 1$
    \item $\lim_{x \to \infty} \left(1 + \dfrac{1}{x}\right)^x = e$
\end{itemize}

\section*{Тригонометрия}
\begin{itemize}
    \item $\sin^2 x + \cos^2 x = 1$
    \item $\sin(2x) = 2 \sin x \cos x$
    \item $\cos(2x) = \cos^2 x - \sin^2 x$
\end{itemize}

\section*{Квадратное уравнение}
Для уравнения $ax^2 + bx + c = 0$:
\begin{itemize}
    \item Дискриминант: $D = b^2 - 4ac$
    \item Корни: $x_{1,2} = \dfrac{-b \pm \sqrt{D}}{2a}$
\end{itemize}
