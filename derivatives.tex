\section*{Таблица производных}

\begin{enumerate}
    \item Производная степенной функции:
    \[
    (x^n)' = n \cdot x^{n-1}
    \]
    
    \item Производная синуса:
    \[
    (\sin x)' = \cos x
    \]
    
    \item Производная косинуса:
    \[
    (\cos x)' = -\sin x
    \]
    
    \item Производная показательной функции (экспоненты):
    \[
    (e^x)' = e^x
    \]
    
    \item Производная натурального логарифма:
    \[
    (\ln x)' = \frac{1}{x}
    \]
\end{enumerate}
