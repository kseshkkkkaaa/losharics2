\section*{Таблица интегралов}

\begin{enumerate}
    \item Интеграл степенной функции ($n \neq -1$):
    \[
    \int x^n \, dx = \frac{x^{n+1}}{n+1} + C
    \]
    
    \item Интеграл обратной пропорциональности:
    \[
    \int \frac{1}{x} \, dx = \ln|x| + C
    \]
    
    \item Интеграл экспоненты:
    \[
    \int e^x \, dx = e^x + C
    \]
    
    \item Интеграл синуса:
    \[
    \int \sin x \, dx = -\cos x + C
    \]
    
    \item Интеграл косинуса:
    \[
    \int \cos x \, dx = \sin x + C
    \]
\end{enumerate}
