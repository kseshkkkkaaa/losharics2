\noindent \textbf{Задача:} Решить систему уравнений
\[
\begin{cases}
x + y = 5 \\
2x - y = 4
\end{cases}
\]

\noindent \textbf{Решение методом сложения:} \\

\noindent Складываем уравнения:
\[
(x + y) + (2x - y) = 5 + 4
\]
\[
3x = 9
\]
\[
x = 3
\]

\noindent Подставляем в первое уравнение:
\[
3 + y = 5
\]
\[
y = 2
\]

\noindent \textbf{Ответ:} $(3, 2)$ \\

\noindent \textbf{Проверка:} \\
$3 + 2 = 5$ \\
$2 \cdot 3 - 2 = 4 \implies 6 - 2 = 4$
