\noindent \textbf{Задача:} Решить уравнение $x^2 - 5x + 6 = 0$ \\

\noindent \textbf{Решение:}

\begin{enumerate}
    \item Находим дискриминант:
    \[
    D = b^2 - 4ac = (-5)^2 - 4 \cdot 1 \cdot 6 = 25 - 24 = 1
    \]
    
    \item Вычисляем корень из дискриминанта:
    \[
    \sqrt{D} = \sqrt{1} = 1
    \]
    
    \item Находим первый корень:
    \[
    x_1 = \frac{5 + 1}{2} = \frac{6}{2} = 3
    \]
    
    \item Находим второй корень:
    \[
    x_2 = \frac{5 - 1}{2} = \frac{4}{2} = 2
    \]
\end{enumerate}

\noindent \textbf{Ответ:} $x_1 = 2$, $x_2 = 3$
